🧠 Machine Learning from Scratch

1. Linear Regression

import numpy as np class LinearRegression: def __init__(self): self.weights = None def fit(self, X, y, lr=0.01, epochs=100): n, m = X.shape self.weights = np.zeros(m) for _ in range(epochs): pred = X @ self.weights gradient = (2/n) * (X.T @ (pred - y)) self.weights -= lr * gradient def predict(self, X): return X @ self.weights # Test X = np.array([[1],[2],[3],[4]]) y = np.array([2,4,6,8]) model = LinearRegression() model.fit(X, y, lr=0.1, epochs=100) print(f"Weights: {model.weights}") print(f"Predictions: {model.predict(X)}")

from qiskit import QuantumCircuit from qiskit.quantum_info import Statevector import numpy as np # Create circuit qc = QuantumCircuit(2) qc.h(0) # Hadamard - superposition qc.cx(0, 1) # CNOT - entanglement # Draw print("Quantum Circuit:") print(qc.draw()) # Execute sv = Statevector.from_int(0, 2**2) sv = sv.evolve(qc) print(f"Statevector: {sv.data.round(3)}") print(f"Probabilities: {sv.probabilities().round(3)}")
